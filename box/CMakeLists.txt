# octavio-box: the Octavio groovebox program for the hardware synth. Builds on its own (no JUCE):
#   cmake -S box -B build-box -DCMAKE_BUILD_TYPE=Release
#   cmake --build build-box
# Mac: CoreAudio + CoreMIDI. Linux / Raspberry Pi: ALSA (sudo apt install libasound2-dev).
# -DOCTAVIO_NO_DEVICES=ON builds only the offline renderer (--render), with no audio library at all.
cmake_minimum_required(VERSION 3.16)
project(OctavioBox CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()
option(OCTAVIO_NO_DEVICES "Offline renderer only" OFF)
option(OCTAVIO_SDL "Screen in a window when SDL2 is found (computers)" ON)
# the screen: drawn in software (ui/), shown through KMS on the Pi or SDL2 on a computer
set(UI_SOURCES ui/Canvas.cpp ui/DisplayDrm.cpp ui/DisplaySdl.cpp ui/Png.cpp)

if(OCTAVIO_NO_DEVICES)
    add_executable(octavio-box main.cpp hw_linux.cpp ${UI_SOURCES})
    target_compile_definitions(octavio-box PRIVATE OCTAVIO_NO_DEVICES)
else()
    add_executable(octavio-box main.cpp hw_linux.cpp ${UI_SOURCES} third_party/rtaudio/RtAudio.cpp third_party/rtmidi/RtMidi.cpp)
    if(OCTAVIO_SDL)
        find_package(SDL2 QUIET)
        if(SDL2_FOUND)
            target_compile_definitions(octavio-box PRIVATE OCTAVIO_SDL)
            if(TARGET SDL2::SDL2)
                target_link_libraries(octavio-box PRIVATE SDL2::SDL2)
            else()
                target_include_directories(octavio-box PRIVATE ${SDL2_INCLUDE_DIRS})
                target_link_libraries(octavio-box PRIVATE ${SDL2_LIBRARIES})
            endif()
            message(STATUS "octavio-box: screen window with SDL2")
        else()
            message(STATUS "octavio-box: SDL2 not found, no screen window (brew install sdl2 / apt install libsdl2-dev)")
        endif()
    endif()
    if(APPLE)
        target_compile_definitions(octavio-box PRIVATE __MACOSX_CORE__)
        target_link_libraries(octavio-box PRIVATE "-framework CoreAudio" "-framework CoreMIDI" "-framework CoreFoundation" "-framework CoreServices")
    elseif(UNIX)
        find_package(ALSA REQUIRED)
        find_package(Threads REQUIRED)
        target_compile_definitions(octavio-box PRIVATE __LINUX_ALSA__)
        target_link_libraries(octavio-box PRIVATE ALSA::ALSA Threads::Threads)
    endif()
endif()
# the screen's fonts (DejaVu Sans Condensed); --fonts overrides
target_compile_definitions(octavio-box PRIVATE OCTAVIO_FONT_DIR="${CMAKE_CURRENT_SOURCE_DIR}/third_party/fonts")
if(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64")
    target_compile_options(octavio-box PRIVATE -O3)
endif()
